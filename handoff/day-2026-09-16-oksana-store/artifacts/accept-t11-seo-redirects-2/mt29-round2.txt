+/**
+ * The storefront locales a tenant's redirect rows are minted for. Falls back
+ * to `["en"]`, the same default a shop with no saved `tenant_locale_settings`
+ * row gets everywhere else (t06-locale-currency): a rename before the owner
+ * ever touches the locale screen still gets a working redirect for the one
+ * language the shop answers in today.
+ */
+export const getTenantEnabledLocales = async (
+  tenantModuleService: TenantModuleService,
+  tenantId: string,
+): Promise<string[]> => {
+  const [settings] = await tenantModuleService.listTenantLocaleSettings({ tenant_id: tenantId });
+  const locales = settings?.enabled_locales as unknown as string[] | undefined;
+  return locales && locales.length > 0 ? locales : ["en"];

+/**
+ * `INVALID_DATA` if `handle` is already used by another product/category —
+ * checked before the update workflow runs, so a collision is one clear
+ * message rather than a database constraint error surfacing through
+ * Medusa's own unique-handle enforcement (step 2 of t11-seo-redirects: "A
+ * handle that collides with an existing one is rejected with a message, not
+ * silently suffixed").
+ */
+export const assertHandleFree = async (
+  req: AuthenticatedMedusaRequest,
+  entity: "product" | "product_category",
+  handle: string,
+  excludingId: string,
+): Promise<void> => {
+  const query = req.scope.resolve(ContainerRegistrationKeys.QUERY);
+
+  const { data } = await query.graph({
+    entity,
+    fields: ["id"],
+    filters: { handle: [handle] },
+  });
+
+  const collision = data.find((row) => row?.id && row.id !== excludingId);
+
+  if (collision) {
+    throw new MedusaError(
+      MedusaError.Types.INVALID_DATA,
+      `Invalid request: the handle "${handle}" is already in use`,
+    );
+  }
+};
+
+/**
+ * Writes (or moves) the 301 for a renamed product/category, one row per the
+ * tenant's enabled locale: `/<locale>/<kind>/<oldHandle>` →
+ * `/<locale>/<kind>/<newHandle>`.
+ *
+ * A rename BACK to a handle this same resource used before would otherwise
+ * leave a stale row pointing the other way and turn the pair into a 2-hop
+ * loop the moment it is re-applied — so the reverse row is dropped first. An
+ * existing forward row (a second rename of the same resource before the first
+ * one's redirect was ever visited) is updated in place rather than left as a
+ * dead extra hop.
+ */
+export const recordSlugRedirect = async (
+  tenantModuleService: TenantModuleService,
+  tenantId: string,
+  kind: "products" | "categories",
+  oldHandle: string,
+  newHandle: string,
+  locales: string[],
+): Promise<void> => {
+  if (oldHandle === newHandle) {
+    return;
+  }
+
+  for (const locale of locales) {
+    const fromPath = `/${locale}/${kind}/${oldHandle}`;
+    const toPath = `/${locale}/${kind}/${newHandle}`;
+
+    if (fromPath === toPath) {
+      continue;
+    }
+
+    const [reverse] = await tenantModuleService.listTenantRedirects({
+      tenant_id: tenantId,
+      from_path: toPath,
+    });
+
+    if (reverse && reverse.to_path === fromPath) {
+      await tenantModuleService.deleteTenantRedirects([reverse.id]);
+    }
+
+    const [existing] = await tenantModuleService.listTenantRedirects({
+      tenant_id: tenantId,
+      from_path: fromPath,
+    });
+
+    if (existing) {
+      await tenantModuleService.updateTenantRedirects({
+        id: existing.id,
+        to_path: toPath,
+        status_code: 301,
+      });
+    } else {
+      await tenantModuleService.createTenantRedirects({
+        tenant_id: tenantId,
+        from_path: fromPath,
+        to_path: toPath,
+        status_code: 301,
+      });
+    }
+  }
+};

+/**
+ * A tenant's path-for-path redirect table (t11-seo-redirects), fetched from
+ * `GET /store/tenant/redirects` and cached in the middleware's own process —
+ * the same shape as `regionMapCache` in `src/middleware.ts`, because
+ * middleware runs in its own runtime with no access to the server components'
+ * Next data cache. Sixty seconds per key: a redirect added in the panel is
+ * live within that window, no restart and no deploy.
+ */
+
+export type TenantRedirect = { from_path: string; to_path: string; status_code: number }
+
+const TTL_MS = 60 * 1000
+const MAX_REDIRECT_HOPS = 20
+
+const redirectCache = new Map<string, { byPath: Map<string, TenantRedirect>; updatedAt: number }>()
+
+async function fetchRedirects(
+  backendUrl: string,
+  publishableKey: string,
+): Promise<Map<string, TenantRedirect>> {
+  const byPath = new Map<string, TenantRedirect>()
+
+  try {
+    const response = await fetch(`${backendUrl}/store/tenant/redirects`, {
+      method: "GET",
+      headers: { "x-publishable-api-key": publishableKey },
+      cache: "no-store",
+    })
+
+    if (!response.ok) {
+      return byPath
+    }
+
+    const { redirects } = (await response.json()) as { redirects: TenantRedirect[] }
+
+    for (const row of redirects ?? []) {
+      byPath.set(row.from_path, row)
+    }
+  } catch {
+    // A backend hiccup answers "no redirects this request" rather than taking
+    // down every page of the shop — the same failure mode `getRegionMap`
+    // chose for a dead regions fetch.
+  }
+
+  return byPath
+}
+
+/** The tenant's redirect table, refetched at most once per `TTL_MS`. */
+export async function getTenantRedirects(
+  backendUrl: string,
+  publishableKey: string | null,
+): Promise<Map<string, TenantRedirect>> {
+  if (!publishableKey) {
+    return new Map()
+  }
+
+  const cached = redirectCache.get(publishableKey)
+
+  if (cached && cached.updatedAt > Date.now() - TTL_MS) {
+    return cached.byPath
+  }
+
+  const byPath = await fetchRedirects(backendUrl, publishableKey)
+  redirectCache.set(publishableKey, { byPath, updatedAt: Date.now() })
+
+  return byPath
+}
+
+export type RedirectResolution =
+  | { kind: "redirect"; to: string; status: number }
+  | { kind: "loop" }
+  | { kind: "none" }
+
+/**
+ * Follows the chain starting at `pathname` up to `MAX_REDIRECT_HOPS` hops,
+ * so a rename that itself got renamed still resolves to the current address
+ * in one round trip to the visitor. A path visited twice in the same chain is
+ * a loop (`a → b`, `b → a`) — the acceptance case this table's own docs
+ * describe — and answers `{ kind: "loop" }` rather than looping forever.
+ */
+export function resolveRedirect(
+  pathname: string,
+  byPath: Map<string, TenantRedirect>,
+): RedirectResolution {
+  if (!byPath.has(pathname)) {
+    return { kind: "none" }
+  }
+
+  const visited = new Set<string>()
+  let current = pathname
+  let last: TenantRedirect | undefined
+
+  for (let hop = 0; hop < MAX_REDIRECT_HOPS; hop++) {
+    if (visited.has(current)) {
+      return { kind: "loop" }
+    }
+    visited.add(current)
+
+    const entry = byPath.get(current)
+    if (!entry) {
+      break
+    }
+    last = entry
+    current = entry.to_path
+  }
+
+  if (byPath.has(current)) {
+    // The chain is still redirecting after MAX_REDIRECT_HOPS: treat it the
+    // same as a detected cycle rather than serving a 21st hop.
+    return { kind: "loop" }
+  }
+
+  if (!last) {
+    return { kind: "none" }
+  }
+
+  return { kind: "redirect", to: current, status: last.status_code }
