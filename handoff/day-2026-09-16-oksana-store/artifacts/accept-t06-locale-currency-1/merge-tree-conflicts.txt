+const ENABLED_LOCALES = ["ru", "en"];

+export default async function backfillTenantLocaleSettings({ container, args }: ExecArgs) {

+  const tenantModuleService: TenantModuleService = container.resolve(TENANT_MODULE);

+      skipped += 1;
+      continue;
+    }
